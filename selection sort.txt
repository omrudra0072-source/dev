
#include <stdio.h>

int main()
{
    int n;

    printf("Enter No. of elements:\n");
    scanf("%d", &n);

    if (n <= 0)
    {
        printf("Please enter a positive number of elements.\n");
        return 1;
    }

    int a[n];

    printf("Enter array elements:\n");
    for (int i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    for (int i = 0; i < n - 1; i++)
    {
        int min_index = i;

        for (int j = i + 1; j < n; j++)
        {
            if (a[j] < a[min_index])
            {
                min_index = j;
            }
        }

        if (min_index != i)
        {
            int temp = a[min_index];
            a[min_index] = a[i];
            a[i] = temp;
        }
    }

    printf("Sorted array elements:\n");
    for (int i = 0; i < n; i++)
    {
        printf("%d ", a[i]);
    }

    printf("\n");

    return 0;
}
